    int find(int x) { return p[x] == x ? x : p[x] = find(p[x]); }   // path compression
bool unite(int a, int b) {
  a = find(a); b = find(b);
  if (a == b) return false;
  if (sz[a] < sz[b]) swap(a, b);   // union by size: hang the smaller tree under the bigger
  p[b] = a; sz[a] += sz[b];
  return true;
}
