Each node stores the aggregate of a range. A query or update touches only O(log n) nodes. Run one and count the highlighted nodes (module 03). Switch to lazy propagation to add a value to a whole range in O(log n): fully covered nodes take a tag instead of updating their children, and tags are pushed down only when a later operation needs to go deeper.

Array (edit values, then Rebuild)

Try: range add [1, 6] += 3, then query [4, 5]. Watch the tags move down only along the query's path. The leaves you didn't touch are still "stale", and that is fine: their ancestors' tags hold the missing amount.

int query(int v, int tl, int tr, int l, int r) {
  if (r < tl || tr < l) return NEUTRAL;      // outside
  if (l <= tl && tr <= r) return t[v];      // fully inside
  int tm = (tl + tr) / 2;                    // partial
  return op(query(2*v,   tl,   tm, l, r),
            query(2*v+1, tm+1, tr, l, r));
}
// lazy range add, range sum
void apply(int v, int tl, int tr, ll x) {
  t[v] += x * (tr - tl + 1);   // min/max: t[v] += x
  lz[v] += x;
}
void push(int v, int tl, int tr) {
  if (!lz[v]) return;
  int tm = (tl + tr) / 2;
  apply(2*v, tl, tm, lz[v]); apply(2*v+1, tm+1, tr, lz[v]);
  lz[v] = 0;
}
void add(int v, int tl, int tr, int l, int r, ll x) {
  if (r < tl || tr < l) return;
  if (l <= tl && tr <= r) { apply(v, tl, tr, x); return; }
  push(v, tl, tr);             // before going down!
  int tm = (tl + tr) / 2;
  add(2*v, tl, tm, l, r, x); add(2*v+1, tm+1, tr, l, r, x);
  t[v] = t[2*v] + t[2*v+1];
}
// query: same shape, call push(v, tl, tr) before recursing.
